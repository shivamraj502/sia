import torch
import torch.nn as nn
from torchvision import datasets, transforms
from torchvision.utils import save_image

# Hyperparameters
z_dim = 64
lr = 2e-4
epochs = 50
batch_size = 64

# Transform
transform = transforms.Compose([
    transforms.ToTensor(),
    transforms.Normalize((0.5,), (0.5,))
])

# Data Loader
loader = torch.utils.data.DataLoader(
    datasets.MNIST(
        ".",
        download=True,
        train=True,
        transform=transform
    ),
    batch_size=batch_size,
    shuffle=True
)

# Generator
class Generator(nn.Module):

    def __init__(self):
        super().__init__()

        self.net = nn.Sequential(
            nn.Linear(z_dim, 512),
            nn.LeakyReLU(0.2),

            nn.Linear(512, 256),
            nn.LeakyReLU(0.2),

            nn.Linear(256, 784),
            nn.Tanh()
        )

    def forward(self, x):
        return self.net(x)

# Discriminator
class Discriminator(nn.Module):

    def __init__(self):
        super().__init__()

        self.net = nn.Sequential(
            nn.Linear(784, 512),
            nn.LeakyReLU(0.2),

            nn.Linear(512, 256),
            nn.LeakyReLU(0.2),

            nn.Linear(256, 1),
            nn.Sigmoid()
        )

    def forward(self, x):
        return self.net(x.view(-1, 784))

# Create Generator and Discriminator
G = Generator()
D = Discriminator()

# Optimizers
opt_G = torch.optim.Adam(G.parameters(), lr=lr)
opt_D = torch.optim.Adam(D.parameters(), lr=lr)

criterion = nn.BCELoss()

# Training
for epoch in range(epochs):

    for real, _ in loader:

        bs = real.size(0)

        # Flatten real images
        real = real.view(bs, -1)

        # Labels
        real_lbl = torch.ones(bs, 1)
        fake_lbl = torch.zeros(bs, 1)

        # Train Discriminator
        z = torch.randn(bs, z_dim)
        fake = G(z).detach()

        d_loss = (
            criterion(D(real), real_lbl)
            + criterion(D(fake), fake_lbl)
        )

        opt_D.zero_grad()
        d_loss.backward()
        opt_D.step()

        # Train Generator
        z = torch.randn(bs, z_dim)
        fake = G(z)

        g_loss = criterion(D(fake), real_lbl)

        opt_G.zero_grad()
        g_loss.backward()
        opt_G.step()

    # Print losses
    print(
        f"Epoch {epoch + 1}/{epochs} "
        f"D_loss: {d_loss.item():.4f} "
        f"G_loss: {g_loss.item():.4f}"
    )

    # Save generated images every 10 epochs
    if (epoch + 1) % 10 == 0:

        with torch.no_grad():
            z = torch.randn(64, z_dim)
            samples = G(z).view(64, 1, 28, 28)

        save_image(
            samples,
            f"sample_epoch_{epoch + 1}.png",
            nrow=8,
            normalize=True
        )

# Save final generated images
with torch.no_grad():
    z = torch.randn(64, z_dim)
    samples = G(z).view(64, 1, 28, 28)

save_image(
    samples,
    "sample.png",
    nrow=8,
    normalize=True
)

print("Generated images saved as sample.png")

import torch
from transformers import AutoTokenizer, AutoModelForCausalLM

name = "Qwen/Qwen2.5-0.5B-Instruct"

tok = AutoTokenizer.from_pretrained(name)
model = AutoModelForCausalLM.from_pretrained(name)

device = "cuda" if torch.cuda.is_available() else "cpu"
model.to(device).eval()

context = (
    "Generative AI creates text and images from learned patterns. "
    "It supports education, design, research, and software development."
)

def ask(question):
    messages = [
        {
            "role": "system",
            "content": "Answer clearly and use only the given context."
        },
        {
            "role": "user",
            "content": f"Context: {context}\n\n{question}"
        }
    ]

    prompt = tok.apply_chat_template(
        messages,
        tokenize=False,
        add_generation_prompt=True
    )

    inputs = tok(prompt, return_tensors="pt").to(device)

    with torch.no_grad():
        output = model.generate(
            **inputs,
            max_new_tokens=100,
            do_sample=False,
            pad_token_id=tok.eos_token_id
        )

    answer = output[0][inputs["input_ids"].shape[1]:]
    return tok.decode(answer, skip_special_tokens=True)

print("Summary:", ask("Summarize the context in one sentence."))
print("Q&A:", ask("What are two uses of generative AI?"))

from datasets import Dataset
from transformers import (
    AutoTokenizer,
    AutoModelForCausalLM,
    Trainer,
    TrainingArguments,
    DataCollatorForLanguageModeling
)
from peft import LoraConfig, get_peft_model, TaskType

name = "facebook/opt-125m"

tok = AutoTokenizer.from_pretrained(name)
tok.pad_token = tok.eos_token

model = AutoModelForCausalLM.from_pretrained(name)

config = LoraConfig(
    task_type=TaskType.CAUSAL_LM,
    r=8,
    lora_alpha=16,
    lora_dropout=0.1,
    target_modules=["q_proj", "v_proj"]
)

model = get_peft_model(model, config)

texts = [
    "Artificial intelligence can generate text from prompts.",
    "LoRA reduces the number of trainable parameters.",
    "Transformers use attention to process sequences.",
    "Generative AI is used in education and research."
]

ds = Dataset.from_dict({"text": texts})

def tokenize(batch):
    return tok(batch["text"], truncation=True, max_length=64)

ds = ds.map(tokenize, batched=True)

args = TrainingArguments(
    output_dir="lora-output",
    per_device_train_batch_size=1,
    num_train_epochs=3,
    logging_steps=1,
    save_strategy="no",
    report_to="none"
)

trainer = Trainer(
    model=model,
    args=args,
    train_dataset=ds,
    data_collator=DataCollatorForLanguageModeling(
        tokenizer=tok,
        mlm=False
    )
)

trainer.train()

model.save_pretrained("opt-lora-adapter")
tok.save_pretrained("opt-lora-adapter")

model.print_trainable_parameters()

import torch
from transformers import AutoTokenizer, AutoModelForSeq2SeqLM
from collections import Counter

name = "google/flan-t5-small"

tok = AutoTokenizer.from_pretrained(name)
model = AutoModelForSeq2SeqLM.from_pretrained(name)

device = "cuda" if torch.cuda.is_available() else "cpu"
model.to(device)
model.eval()

prompt = (
    "Solve carefully and give only the final numeric answer. "
    "A train travels at 60 km/h for 2 hours. How many km?"
)

inputs = tok(prompt, return_tensors="pt").to(device)

answers = []

for _ in range(5):
    with torch.no_grad():
        output = model.generate(
            **inputs,
            max_new_tokens=40,
            do_sample=True,
            temperature=0.7
        )

    answers.append(tok.decode(output[0], skip_special_tokens=True))

print("Samples:", answers)
print("Most common answer:", Counter(answers).most_common(1)[0][0])
